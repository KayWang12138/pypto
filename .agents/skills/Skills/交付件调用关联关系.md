### 2.1 Operation层 → Function层

**调用关系**:
```cpp
// interface/operation/vector/binary.cpp
LogicalTensorPtr Add(Function &function, const TileShape &tileShape, 
                     const LogicalTensorPtr &input1, const LogicalTensorPtr &input2) {
    // 1. 创建输出Tensor
    auto result = std::make_shared<LogicalTensor>(function, dtype, resultShape);
    
    // 2. 切分Tile并添加Operation
    TiledBinaryOperation<BinaryOpType::ADD>(function, tileShape, 0, 
                                            input1, input2, result, resultTileInfo);
    
    return result;
}

template <BinaryOpType T>
void TiledBinaryOperation(Function &function, ...) {
    // 递归切分Tile
    if (cur == input1.tensor->GetShape().size()) {
        // 创建Tile视图
        auto inputTile1 = input1.tensor->View(function, input1.tileInfo.shape, input1.tileInfo.offset);
        auto inputTile2 = input2.tensor->View(function, input2.tileInfo.shape, input2.tileInfo.offset);
        auto resultTile = result->View(function, resultTileInfo.shape, resultTileInfo.offset);
        
        // 添加Operation到Function
        function.AddOperation(
            Opcode::OP_ADD,           // 操作码
            {inputTile1, inputTile2}, // 输入Tile
            {resultTile}              // 输出Tile
        );
        return;
    }
    // 继续切分...
}
```

**关键数据流**:
- **输入**: LogicalTensor（完整Tensor）
- **输出**: LogicalTensor（完整Tensor）
- **中间产物**: Operation对象（包含Tile视图）

**文件关联**:
```
interface/operation/vector/binary.cpp
    ├─> interface/function/function.h (AddOperation)
    ├─> interface/tensor/logical_tensor.h (View)
    └─> interface/operation/operation.h (Operation构造)
```

```cpp
// codegen/cloudnpu/codegen_vector_binary.cpp
std::string CodeGenOpCloudNPU::PrintBinaryStatic(const PrintBinaryParam &param) const {
    // 1. 准备模板参数
    std::vector<std::string> paramList;
    paramList.emplace_back(dstDtypeStr);  // 数据类型
    paramList.emplace_back("/*OS0*/");    // OriginShape

        paramList.emplace_back(std::to_string(os0[i]));
    }
    // ... 更多Shape参数
    
    // 2. 准备函数调用参数
    std::string dst = "(__ubuf__ " + dstDtypeStr + "*)" + dVar;
    std::string src0 = "(__ubuf__ " + src0DtypeStr + "*)" + s0Var;
    std::string src1 = "(__ubuf__ " + src1DtypeStr + "*)" + s1Var;
    
    // 3. 生成TileOp调用代码
    std::ostringstream os;
    os << "Binary_<" << JoinString(paramList, ",") << ">"
       << "(" << dst << ", " << src0 << ", " << src1 << ");\n";
    
    return os.str();
}
```

### 2.4 TileOp层 → PTO指令层

**调用关系**:
```cpp
// interface/tileop/vector/binary.h
template <BinaryOp op, TileOp::BroadcastOperand operand, typename LastUse, 
          typename T0, typename T1, typename T2>
TILEOP void BinaryComputeImpl(T0 dst, T1 src0, T2 src1) {
    constexpr auto n1 = Std::tuple_element<DIM_1ST, LastUse>::type::value;
    constexpr auto n2 = Std::tuple_element<DIM_2ND, LastUse>::type::value;
    constexpr auto n3 = Std::tuple_element<DIM_3RD, LastUse>::type::value;
    
    if constexpr (op == BinaryOp::ADD) {
        if constexpr (operand == TileOp::BroadcastOperand::NONE) {
            // 调用PTO指令
            PTO_WITH_LAST_USE(pto::TADD(dst, src0, src1), n1, n2, n3);
        } else {
            // 调用广播PTO指令
            PTO_WITH_LAST_USE(pto::TROWEXPANDADD(dst, src0, src1), n1, n2, n3);
        }
    }
    // ... 其他操作
}
```
